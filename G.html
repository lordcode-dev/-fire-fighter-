💻 Stage 1 Arduino C++

/*
  Eaii Firefighter Robot
  STAGE 1 — Motor Movement Test

  Goal:
  Forward -> stop -> backward -> stop

  4WD:
  Motor A = left side
  Motor B = right side
*/

const int ENA = 5;

const int IN1 = 7;
const int IN2 = 8;

const int IN3 = 9;
const int IN4 = 10;

void stopRobot() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}

void forward() {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}

void backward() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}

void setup() {

  pinMode(ENA, OUTPUT);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);

  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  analogWrite(ENA, 180);

  stopRobot();
}

void loop() {

  forward();
  delay(3000);

  stopRobot();
  delay(1500);

  backward();
  delay(3000);

  stopRobot();
  delay(3000);
}

💻 Stage 2 Arduino C++

/*
  STAGE 2
  Find flame -> move toward it -> stop at ~10 cm -> LED ON

  IMPORTANT:
  Flame sensor modules can have different output behavior.
  Adjust FLAME_THRESHOLD after testing your actual modules.
*/

const int ENA = 5;

const int IN1 = 7;
const int IN2 = 8;
const int IN3 = 9;
const int IN4 = 10;

const int FLAME_LEFT   = A0;
const int FLAME_CENTER = A1;
const int FLAME_RIGHT  = A2;

const int TRIG = 11;
const int ECHO = 12;

const int LED = 13;

const int STOP_DISTANCE = 10;

int readDistance() {

  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);

  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);

  digitalWrite(TRIG, LOW);

  long duration = pulseIn(ECHO, HIGH, 30000);

  if (duration == 0) return 999;

  return duration * 0.0343 / 2;
}

void stopRobot() {

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}

void forward() {

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}

void turnLeft() {

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}

void turnRight() {

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}

void setup() {

  pinMode(ENA, OUTPUT);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);

  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);

  pinMode(LED, OUTPUT);

  analogWrite(ENA, 170);

  stopRobot();
}

void loop() {

  int left   = analogRead(FLAME_LEFT);
  int center = analogRead(FLAME_CENTER);
  int right  = analogRead(FLAME_RIGHT);

  int distance = readDistance();

  /*
    You MUST test whether your flame sensor value
    becomes higher or lower when it sees fire.

    This example assumes LOWER = stronger flame.
  */

  if (distance <= STOP_DISTANCE) {

    stopRobot();

    digitalWrite(LED, HIGH);

    while (true) {
      // Wait for Stage 3.
    }
  }

  if (center < left && center < right) {

    forward();

  } else if (left < right) {

    turnLeft();

  } else {

    turnRight();
  }

  delay(40);
}

💻 Stage 3 Arduino C++

#include <Servo.h>

/*
  STAGE 3 — WATER ATTACK

  Relay controls the water pump.
  Servo rotates the nozzle.

  IMPORTANT:
  Relay modules may be active HIGH or active LOW.
  Test your relay before running the pump.
*/

const int RELAY_PIN = 4;
const int LED = 13;

const int SERVO_PIN = 6;

Servo nozzle;

void pumpOn() {
  digitalWrite(RELAY_PIN, LOW);
}

void pumpOff() {
  digitalWrite(RELAY_PIN, HIGH);
}

void setup() {

  pinMode(RELAY_PIN, OUTPUT);
  pinMode(LED, OUTPUT);

  nozzle.attach(SERVO_PIN);

  pumpOff();

  nozzle.write(90);

  digitalWrite(LED, HIGH);
}

void loop() {

  // Aim left
  nozzle.write(60);
  delay(500);

  pumpOn();
  delay(1200);

  pumpOff();

  // Aim center
  nozzle.write(90);
  delay(400);

  pumpOn();
  delay(1200);

  pumpOff();

  // Aim right
  nozzle.write(120);
  delay(500);

  pumpOn();
  delay(1200);

  pumpOff();

  // Return center
  nozzle.write(90);

  delay(3000);
}
⚠️ Keep water away from the Arduino, L298N, relay signal wiring and exposed electrical connections. The pump's electrical supply must also match the actual pump specification.

Navigation

Previous progress — Stage 3

Navigation Sensors

Ultrasonic Sensor 3 Flame Sensors 4WD Motors L298N

Navigation Logic

  1. Move forward.
  2. Check ultrasonic distance.
  3. If obstacle is close, stop.
  4. Turn to find a clear path.
  5. Continue searching.
  6. Use flame sensors to locate fire.

💻 Stage 5 Arduino C++

/*
  STAGE 5 — LONG RANGE FIRE SEARCH

  Target:
  Detect a flame that may be farther than 1 meter.

  NOTE:
  There is no universal analog value that means
  "fire is exactly 1 meter away."

  You MUST calibrate FLAME_THRESHOLD with your
  actual flame sensor modules.
*/

const int ENA = 5;

const int IN1 = 7;
const int IN2 = 8;
const int IN3 = 9;
const int IN4 = 10;

const int FLAME_LEFT   = A0;
const int FLAME_CENTER = A1;
const int FLAME_RIGHT  = A2;

const int TRIG = 11;
const int ECHO = 12;

const int FLAME_THRESHOLD = 500;

int distanceCM() {

  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);

  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);

  digitalWrite(TRIG, LOW);

  long duration = pulseIn(ECHO, HIGH, 30000);

  if (duration == 0) return 999;

  return duration * 0.0343 / 2;
}

void stopRobot() {

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}

void forward() {

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}

void left() {

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}

void right() {

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}

void setup() {

  pinMode(ENA, OUTPUT);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);

  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);

  Serial.begin(9600);

  analogWrite(ENA, 160);

  stopRobot();
}

void loop() {

  int leftValue   = analogRead(FLAME_LEFT);
  int centerValue = analogRead(FLAME_CENTER);
  int rightValue  = analogRead(FLAME_RIGHT);

  int distance = distanceCM();

  Serial.print("Left: ");
  Serial.print(leftValue);

  Serial.print(" Center: ");
  Serial.print(centerValue);

  Serial.print(" Right: ");
  Serial.print(rightValue);

  Serial.print(" Distance: ");
  Serial.println(distance);

  /*
    Example assumes lower analog value = stronger flame.

    If your sensor works opposite:
    change < to >.
  */

  bool flameDetected =
    leftValue < FLAME_THRESHOLD ||
    centerValue < FLAME_THRESHOLD ||
    rightValue < FLAME_THRESHOLD;

  if (!flameDetected) {

    // Slowly scan for fire.
    left();
    delay(250);

    stopRobot();
    delay(100);

    continue;
  }

  /*
    Fire direction found.
  */

  if (centerValue <= leftValue &&
      centerValue <= rightValue) {

    forward();

  } else if (leftValue < rightValue) {

    left();

  } else {

    right();
  }

  delay(100);
}
